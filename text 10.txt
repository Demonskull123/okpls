			if d:IsA("ProximityPrompt") then prompt = d break end
		end
	end
	if prompt and prompt:IsA("ProximityPrompt") then
		fireproximityprompt(prompt)
		return true
	end
	return false
end

local function FarmTpSelectGenerator()
	local localChar = Players.LocalPlayer.Character
	local root = localChar and localChar:FindFirstChild("HumanoidRootPart")
	if not root then return nil end
	local best, bestDist = nil, math.huge
	for _, gen in pairs(FarmGetGenerators()) do
		if FarmGetProgress(gen) < 100 then
			local pos = FarmGetGeneratorPos(gen)
			if pos ~= Vector3.new(0, 0, 0) then
				local d = (pos - root.Position).Magnitude
				if d < bestDist then bestDist = d best = gen end
			end
		end
	end
	return best
end

local function DhGetPing()
	local ok, v = pcall(function()
		return game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
	end)
	return (ok and v) or 0.08
end

local function DhIsBeheading(model)
	if not t3.ignoreBehead or not model then return false end
	if beheadingTargets[model] then return true end
	if model:FindFirstChild("BeheadAbility") then return true end
	for _, c in ipairs(model:GetChildren()) do
		local n = string.lower(c.Name)
		if n == "behead" or string.find(n, "behead_") then return true end
	end
	return false
end

local function DhIsBot(nameLower)
	if string.find(nameLower, "bot") or string.find(nameLower, "minion")
		or string.find(nameLower, "pizza") or string.find(nameLower, "corrupt") then
		return true
	end
	return false
end

local function DhFireBlock()
	if not t1.autoBlockNewActive then return end
	local now = tick()
	if (now - lastDhBlockFire) < (t3.blockDelay or 0) then return end
	lastDhBlockFire = now
	local remote = game.ReplicatedStorage.Modules.Network.RemoteEvent
	pcall(function() remote:FireServer("UseActorAbility", "Block") end)
end

local function DhPredictBoxCFrame(targetRoot)
	local ping = DhGetPing()
	local pingStuds = ping * 22
	local depth = t3.boxSizeDepth + t3.hitboxDistance + 3 + pingStuds
	local offsetZ = -(t3.boxSizeDepth / 2) - t3.hitboxDistance - pingStuds
	return targetRoot.CFrame * CFrame.new(0, 0, offsetZ), depth
end

local function DhBuildBox(target)
	if blockBoxes[target] then return end
	local root = target:FindFirstChild("HumanoidRootPart")
	if not root then return end
	local size = Vector3.new(t3.boxSizeWidth, t3.boxSizeHeight, t3.boxSizeDepth)
	local part = Instance.new("Part")
	part.Name = "SNT_HybridBox"
	part.Size = size
	part.Color = t3.boxColor
	part.Transparency = t1.blockOnHitbox and t3.boxTransparency or 1
	part.Material = t3.glowMode and Enum.Material.Neon or Enum.Material.SmoothPlastic
	part.CanCollide = false
	part.CanQuery = false
	part.Massless = true
	part.Anchored = true
	part.Parent = workspace
	blockBoxes[target] = part
	local born = os.clock()
	local fired = false
	local params = OverlapParams.new()
	params.FilterType = Enum.RaycastFilterType.Include
	local conn
	conn = RunService.Heartbeat:Connect(function()
		if not part.Parent then
			conn:Disconnect() blockBoxes[target] = nil blockBoxConns[target] = nil return
		end
		local r = target:FindFirstChild("HumanoidRootPart")
		if not r then
			conn:Disconnect() part:Destroy() blockBoxes[target] = nil blockBoxConns[target] = nil return
		end
		local cf = DhPredictBoxCFrame(r)
		part.CFrame = cf
		local elapsed = os.clock() - born
		if elapsed > (t3.visibleTime or 0.4) or DhIsBeheading(target) then
			conn:Disconnect() part:Destroy() blockBoxes[target] = nil blockBoxConns[target] = nil return
		end
		if not fired and t1.autoBlockNewActive then
			local myChar = Players.LocalPlayer.Character
			if myChar then
				params.FilterDescendantsInstances = { myChar }
				local hit = workspace:GetPartBoundsInBox(cf, size, params)
				if #hit > 0 then
					fired = true
					part.Color = Color3.fromRGB(255, 0, 0)
					DhFireBlock()
				end
			end
		end
	end)
	blockBoxConns[target] = conn
end

local function DhHookSound(snd)
	if not snd:IsA("Sound") or hookedSounds[snd] then return end
	hookedSounds[snd] = true
	local function onFire()
		if not t1.autoBlockNewActive then return end
		local sid = snd.SoundId
		if not sid or sid == "" then return end
		local num = string.match(sid, "%d+$")
		if not num then return end
		local model = snd:FindFirstAncestorOfClass("Model")
		if not model or model == Players.LocalPlayer.Character then return end
		local n = string.lower(model.Name)
		if DhIsBot(n) then return end
		local isBehead = beheadAnimIds[num]
		local isM1 = blockAnimIds[num]
		if isBehead and t3.ignoreBehead then
			beheadingTargets[model] = true
			if blockBoxes[model] then blockBoxes[model]:Destroy() blockBoxes[model] = nil end
			task.delay(2.5, function() beheadingTargets[model] = nil end)
			return
		end
		if not isM1 then return end
		local myChar = Players.LocalPlayer.Character
		local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
		local theirRoot = model:FindFirstChild("HumanoidRootPart")
		if myRoot and theirRoot then
			local ping = DhGetPing()
			local pingStuds = ping * 22
			local depth = t3.boxSizeDepth + t3.hitboxDistance + 3 + pingStuds
			local halfX = t3.boxSizeWidth / 2
			local halfY = t3.boxSizeHeight / 2
			local halfZ = depth / 2
			local offsetZ = -(t3.boxSizeDepth / 2) - t3.hitboxDistance - pingStuds
			local boxCF = theirRoot.CFrame * CFrame.new(0, 0, offsetZ)
			local localPos = boxCF:ToObjectSpace(myRoot.CFrame).Position
			if math.abs(localPos.X) <= halfX and math.abs(localPos.Y) <= halfY and math.abs(localPos.Z) <= halfZ then
				DhFireBlock()
			end
		end
		DhBuildBox(model)
	end
	snd.Played:Connect(onFire)
	snd.Resumed:Connect(onFire)
	snd:GetPropertyChangedSignal("Playing"):Connect(function()
		if snd.IsPlaying then onFire() end
	end)
	if snd.IsPlaying then onFire() end
end

local function DhScanSounds()
	for _, d in ipairs(workspace:GetDescendants()) do
		if d:IsA("Sound") then DhHookSound(d) end
	end
	if not soundScanConn then
		soundScanConn = workspace.DescendantAdded:Connect(function(d)
			if d:IsA("Sound") then DhHookSound(d) end
		end)
	end
end

local function DhHookAnimator(character)
	if not character then return end
	task.spawn(function()
		local h = character:WaitForChild("Humanoid", 3)
		if not h then return end
		local anim = h:WaitForChild("Animator", 3)
		if not anim then return end
		anim.AnimationPlayed:Connect(function(track)
			if not t1.autoBlockNewActive or not t3.ignoreBehead then return end
			if track.Animation then
				local id = string.match(track.Animation.AnimationId, "%d+")
				if id and beheadAnimIds[id] then
					beheadingTargets[character] = true
					task.delay(track.Length > 0 and track.Length or 2.5, function()
						beheadingTargets[character] = nil
					end)
				end
			end
		end)
	end)
end

local function DhHookAllPlayers()
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr.Character then DhHookAnimator(plr.Character) end
		plr.CharacterAdded:Connect(DhHookAnimator)
	end
	Players.PlayerAdded:Connect(function(p)
		p.CharacterAdded:Connect(DhHookAnimator)
	end)
end

	if t1.autoBlockNewActive then return end
	t1.autoBlockNewActive = true
	DhScanSounds()
	DhHookAllPlayers()
end

	for target, part in pairs(blockBoxes) do if part then part:Destroy() end end
	blockBoxes = {}
	for target, c in pairs(blockBoxConns) do if c then c:Disconnect() end end
	blockBoxConns = {}
end

local function M4AutoBlockLoop(mode)
	local LocalPlayer = Players.LocalPlayer
	local Killers = workspace:WaitForChild("Players"):WaitForChild("Killers")
	while t1.m4AutoBlockSoundEnabled or t1.m4AutoBlockAnimEnabled do
		local Character = LocalPlayer.Character
		local myRoot = Character and Character:FindFirstChild("HumanoidRootPart")
		if myRoot then
			for _, killer in ipairs(Killers:GetChildren()) do
				if killer:IsA("Model") then
					local theirRoot = killer:FindFirstChild("HumanoidRootPart")
					if theirRoot then
						local inRange = (myRoot.Position - theirRoot.Position).Magnitude <= t3.m4BlockRange
						local facingOK = true
						if t3.m4BlockFacing and inRange then
							local unit = (myRoot.Position - theirRoot.Position).Unit
							facingOK = theirRoot.CFrame.LookVector:Dot(unit) > t3.m4BlockDot
						end
						if mode == "Sound" and t1.m4AutoBlockSoundEnabled then
							for _, d in ipairs(killer:GetDescendants()) do
								if d:IsA("Sound") and m4BlockSoundIds[d.SoundId] and d.IsPlaying then
									if m4SoundState[d] == nil then m4SoundState[d] = not inRange end
									local shouldFire = inRange and facingOK and not m4SoundState[d] and not m4SoundFired[d]
									if shouldFire then
										m4SoundFired[d] = true
										M4FireBlock()
									end
								elseif d:IsA("Sound") and not d.IsPlaying then
									m4SoundState[d] = nil
									m4SoundFired[d] = nil
								end
							end
						elseif mode == "Animation" and t1.m4AutoBlockAnimEnabled then
							local h = killer:FindFirstChildOfClass("Humanoid")
							local anim = h and h:FindFirstChildOfClass("Animator")
							if anim then
								for _, track in ipairs(anim:GetPlayingAnimationTracks()) do
									local a = track.Animation
									local id = a and a.AnimationId
									if id and m4BlockAnimIds[id] then
										if m4SoundState[track] == nil then m4SoundState[track] = not inRange end
										local shouldFire = inRange and facingOK and not m4SoundState[track] and not m4SoundFired[track]
										if shouldFire then
											m4SoundFired[track] = true
											M4FireBlock()
										end
									end
								end
								for k in pairs(m4SoundState) do
									if typeof(k) == "Instance" and k:IsA("AnimationTrack") and not k.IsPlaying then
										m4SoundState[k] = nil
										m4SoundFired[k] = nil
									end
								end
							end
						end
					end
				end
			end
		end
		task.wait(0.1)
	end
end

local function M4AutoBlockStart()
	if t4.m4SoundBlockConn then return end
	t4.m4SoundBlockConn = task.spawn(function() M4AutoBlockLoop(t3.m4BlockType) end)

local skillAimbotState = {
	chance = false, twoTime = false, veronica = false, n7 = false, autoClone = false
}

local function ChanceAimbotLoop()
	local LocalPlayer = Players.LocalPlayer
	while skillAimbotState.chance do
		local char = LocalPlayer.Character
		local myRoot = char and char:FindFirstChild("HumanoidRootPart")
		local humanoid = char and char:FindFirstChildOfClass("Humanoid")
		if myRoot and humanoid then
			local killers = workspace:FindFirstChild("Players") and workspace.Players:FindFirstChild("Killers")
			local killer = killers and killers:GetChildren()[1]
			if killer and killer:IsA("Model") then
				local killerRoot = killer:FindFirstChild("HumanoidRootPart") or killer.PrimaryPart
				if killerRoot then
					local origAuto = humanoid.AutoRotate
					humanoid.AutoRotate = false
					local start = tick()
					while tick() - start < 2 and skillAimbotState.chance do
						if not myRoot.Parent or not killerRoot.Parent then break end
						local ahead = killerRoot.Position + killerRoot.CFrame.LookVector * (tonumber(t3.aimbotRange) or 4)
						local flat = Vector3.new(ahead.X, myRoot.Position.Y, ahead.Z)
						myRoot.CFrame = CFrame.lookAt(myRoot.Position, flat)
						RunService.RenderStepped:Wait()
					end
					humanoid.AutoRotate = origAuto
				end
			end
		end
		task.wait(0.1)
	end
end

local function TwoTimeBackstabFire()
	local LocalPlayer = Players.LocalPlayer
	local char = LocalPlayer.Character
	if not char then return end
	local killers = workspace:FindFirstChild("Players") and workspace.Players:FindFirstChild("Killers")
	local killer = killers and killers:GetChildren()[1]
	if not killer or not killer:IsA("Model") then return end
	local killerRoot = killer:FindFirstChild("HumanoidRootPart") or killer.PrimaryPart
	if not killerRoot then return end
	local origPrimary = char.PrimaryPart
	local part = Instance.new("Part")
	part.Name = "BackstabPrimaryPart"
	part.Size = Vector3.new(1, 1, 1)
	part.Transparency = 1
	part.Anchored = true
	part.CanCollide = false
	part.CanTouch = false
	part.CanQuery = false
	part.CFrame = killerRoot.CFrame - killerRoot.CFrame.LookVector * 2
	part.Parent = char
	char.PrimaryPart = part
	local start = tick()
	while tick() - start < 0.5 and skillAimbotState.twoTime do
		if not killerRoot.Parent then break end
		part.CFrame = killerRoot.CFrame - killerRoot.CFrame.LookVector * 2
		RunService.RenderStepped:Wait()
	end
	if part.Parent then part:Destroy() end
	if origPrimary and origPrimary.Parent then char.PrimaryPart = origPrimary end
end

local m4M1Names = { Slash = true, Stab = true, ["Carving Slash"] = true, Punch = true }

local function M4M1AimFire()
	local LocalPlayer = Players.LocalPlayer
	local char = LocalPlayer.Character
	if not char then return end
	local myRoot = char:FindFirstChild("HumanoidRootPart")
	if not myRoot then return end
	local nearest, dist = nil, math.huge
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer and p.Character then
			local hrp = p.Character:FindFirstChild("HumanoidRootPart")
			if hrp then
				local d = (myRoot.Position - hrp.Position).Magnitude
				if d < dist then dist = d nearest = hrp end
			end
		end
	end
	if not nearest then return end
	local start = tick()
	local conn
	conn = RunService.RenderStepped:Connect(function()
		if tick() - start >= 1 or not nearest.Parent or not myRoot.Parent then
			conn:Disconnect()
			return
		end
		local ahead = nearest.Position + nearest.CFrame.LookVector * 4
		local flat = Vector3.new(ahead.X, myRoot.Position.Y, ahead.Z)
		myRoot.CFrame = CFrame.lookAt(myRoot.Position, flat)
	end)
end

local function M4ExtendHitbox(range, duration)

	local start = tick()
	repeat
		RunService.Heartbeat:Wait()
		local char = Players.LocalPlayer.Character
		if char then
			local hrp = char:FindFirstChild("HumanoidRootPart")
			if hrp then
				local origVel = hrp.Velocity
				hrp.Velocity = origVel + hrp.CFrame.LookVector * studs
				RunService.RenderStepped:Wait()
				local c2 = Players.LocalPlayer.Character
				if c2 then
					local hrp2 = c2:FindFirstChild("HumanoidRootPart")
					if hrp2 then hrp2.Velocity = origVel end
				end
			end
		end
	until tick() - start > duration or not t1.m4HitboxExpanderEnabled
end

local function ScanItems()

local distanceConnection = nil
local function StartDistanceUpdater()
	if distanceConnection then return end
	distanceConnection = RunService.Heartbeat:Connect(function()
		local now = tick()
		if now - espState.lastDistanceUpdate < t3.espDistanceUpdateInterval then return end
		espState.lastDistanceUpdate = now
		local root = GetLocalRoot()
		if not root then return end
		local rp = root.Position
		local maxD = t3.espMaxDistance
		if u8.SurvivorsDistance or u8.SurvivorsNameESP then
			for inst, entry in pairs(espState.survivorMap) do
				if entry.billboard and entry.billboard.Parent and entry.head and entry.head.Parent then
					local d = math.floor((entry.head.Position - rp).Magnitude)
					local l = entry.billboard:FindFirstChild("TextLabel")
					if l then
						if d > maxD then
							if l.TextTransparency ~= 1 then l.TextTransparency = 1 end
						else
							local uname = inst:GetAttribute("Username") or inst.Name
							local base = u8.SurvivorsNameESP and uname or ""
							local txt = u8.SurvivorsDistance and (base .. " [" .. d .. "m]") or base

						end
					end
				end
			end
		end
		if u8.KillerDistance or u8.KillerNameESP then
			for inst, entry in pairs(espState.killerMap) do
				if entry.billboard and entry.billboard.Parent and entry.head and entry.head.Parent then
					local d = math.floor((entry.head.Position - rp).Magnitude)
					local l = entry.billboard:FindFirstChild("TextLabel")
					if l then
						if d > maxD then
							if l.TextTransparency ~= 1 then l.TextTransparency = 1 end

							local uname = inst:GetAttribute("Username") or inst.Name
							local base = u8.KillerNameESP and uname or ""

					end
				end
			end
		end

			for inst, entry in pairs(espState.itemMap) do
				if entry.billboard and entry.billboard.Parent and entry.handle and entry.handle.Parent then
					local d = math.floor((entry.handle.Position - rp).Magnitude)
					local l = entry.billboard:FindFirstChild("TextLabel")
					if l then
						if d > maxD then
							if l.TextTransparency ~= 1 then l.TextTransparency = 1 end
						else
							local base = u8.ItemsNameESP and inst.Name or ""

				if entry.billboard and entry.billboard.Parent and entry.part and entry.part.Parent then
					local d = math.floor((entry.part.Position - rp).Magnitude)

function cleanupAllEsp()
	for _, map in pairs(espState) do

					if entry.billboard then pcall(function() entry.billboard:Destroy() end) end
				end
				map[inst] = nil
			end
		end
	end
	if distanceConnection then
		distanceConnection:Disconnect()
		distanceConnection = nil
	end
end

local function u65(p37, p38)
	local t14 = {}
	local LocalCharacter = Players.LocalPlayer.Character
	local myRoot = LocalCharacter and LocalCharacter:FindFirstChild("HumanoidRootPart")
	local maxRange = t3.aimbotRange or 200
	local function validTarget(m)
		if not m or not m.Parent then return false end
		local h = m:FindFirstChildOfClass("Humanoid")
		if not h or h.Health <= 0 then return false end
		local hrp = m:FindFirstChild("HumanoidRootPart")
		if not hrp then return false end
		if myRoot and (hrp.Position - myRoot.Position).Magnitude > maxRange then return false end
		return true
	end
	if not p37 then
		if not p38 then
			local Survivors = workspace.Players:FindFirstChild("Survivors")
			if Survivors then
				for _, child in pairs(Survivors:GetChildren()) do
					if validTarget(child) then table.insert(t14, child) end
				end
			end
		else
			local Killers = workspace.Players:FindFirstChild("Killers")
			if Killers then
				for _, child in pairs(Killers:GetChildren()) do
					if validTarget(child) then table.insert(t14, child) end
				end
			end
		end
	else
		local Survivors = workspace.Players:FindFirstChild("Survivors")
		if Survivors then
			for _, child in pairs(Survivors:GetChildren()) do
				if validTarget(child) then table.insert(t14, child) end
			end
		end
	end
	if #t14 == 0 then return nil end
	if not myRoot then return t14[1] end
	local best, bestDist = nil, math.huge
	for _, v in pairs(t14) do
		local hrp = v:FindFirstChild("HumanoidRootPart")
		if hrp then
			local d = (hrp.Position - myRoot.Position).Magnitude
			if d < bestDist then bestDist = d best = v end
		end
	end
	return best
end

local function StopAimbotLock()
	t1.aimbotLocking = false
	if t4.aimbotLockConnection then
		pcall(function() t4.aimbotLockConnection:Disconnect() end)
		t4.aimbotLockConnection = nil
	end
end

local function StartAimbotLock(initialTarget, isKillerTarget)
	if t4.aimbotLockConnection then
		pcall(function() t4.aimbotLockConnection:Disconnect() end)
		t4.aimbotLockConnection = nil
	end
	t1.aimbotLocking = true
	t2.lastAimLockTime = tick()
	local endTime = tick() + (t3.aimbotLockDuration or 1.5)
	local currentTarget = initialTarget
	t4.aimbotLockConnection = RunService.Heartbeat:Connect(function()
		if not t1.aimbotLocking then StopAimbotLock() return end
		if tick() >= endTime then StopAimbotLock() return end
		local valid = false
		if currentTarget and currentTarget.Parent then
			local h = currentTarget:FindFirstChildOfClass("Humanoid")
			local hrp = currentTarget:FindFirstChild("HumanoidRootPart")
			local myRoot = Players.LocalPlayer.Character and Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
			if h and h.Health > 0 and hrp and myRoot and (hrp.Position - myRoot.Position).Magnitude <= (t3.aimbotRange or 200) then
				valid = true
			end
		end
		if not valid then
			currentTarget = u65(not isKillerTarget, isKillerTarget)
			if not currentTarget then StopAimbotLock() return end
		end
		local targetRoot = currentTarget:FindFirstChild("HumanoidRootPart")
		if not targetRoot then StopAimbotLock() return end
		local myChar = Players.LocalPlayer.Character
		if not myChar then StopAimbotLock() return end
		local myRoot = myChar:FindFirstChild("HumanoidRootPart")
		if not myRoot then StopAimbotLock() return end
		local cam = workspace.CurrentCamera
		if not cam then return end
		local aimPos = targetRoot.Position
		if t3.aimbotUseCamera then
			if t3.aimbotSmooth then
				local targetCF = CFrame.new(cam.CFrame.Position, aimPos)

			local best, dist = nil, math.huge
			local Survivors = workspace.Players:FindFirstChild("Survivors")
			if Survivors then
				for _, s in pairs(Survivors:GetChildren()) do
					if s:IsA("Model") and s:FindFirstChild("Humanoid") and s.Humanoid.Health > 0 then
						local hrp = s:FindFirstChild("HumanoidRootPart")
						if hrp then
							local d = (myRoot.Position - hrp.Position).Magnitude
							if d < dist then dist = d best = s end
						end
					end
				end
			end
			if best and dist <= t3.auraRange then
				if tick() - (t2.lastAuraUse or 0) >= t3.auraCooldown then
					t2.lastAuraUse = tick()
					local name = c.Name
					local ability = "Slash"
					if name == "Noli" then ability = "Stab"
					elseif name == "c001kidd" then ability = "Punch"
					elseif name == "Guest1337" then ability = "Punch" end
					pcall(function()
						game.ReplicatedStorage.Modules.Network.RemoteEvent:FireServer("UseActorAbility", ability)
					end)
				end
			end
		end
	end
end)
